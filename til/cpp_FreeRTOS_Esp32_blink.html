// Define the onboard LED pin for ESP32 (typically GPIO 2)
#define LED_PIN 2

// Task handle
TaskHandle_t ledTaskHandle = NULL;

void ledTask(void *pvParameters) {
  // Retrieve the core number the task is running on
  int core = xPortGetCoreID();

  for (;;) {
    // Toggle the LED
    digitalWrite(LED_PIN, HIGH);
    Serial.printf("LED ON  - Running on core %d\n\r", core);
    vTaskDelay(100 / portTICK_PERIOD_MS);

    digitalWrite(LED_PIN, LOW);
    Serial.printf("LED OFF - Running on core %d\n\r", core);
    vTaskDelay(900 / portTICK_PERIOD_MS);
  }
}

void setup() {
  // Initialize serial communication
  Serial.begin(115200);

  // Initialize the onboard LED pin
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  TaskFunction_t   pvTaskCode         = ledTask;
  const char      *constpcName        = "LED Task";
  const uint32_t   usStackDepth       = 2048;
  void            *constpvParameters  = NULL;
  UBaseType_t      uxPriority         = 1;
  TaskHandle_t    *constpvCreatedTask = &ledTaskHandle;
  const BaseType_t xCoreID            = 1;

  // Create the LED task
  xTaskCreatePinnedToCore(pvTaskCode,         // Task function
                          constpcName,        // Task name
                          usStackDepth,       // Stack size
                          constpvParameters,  // Task parameters
                          uxPriority,         // Task priority
                          constpvCreatedTask, // Task handle
                          xCoreID             // Core number to run the task on (0 or 1)
  );
}

void loop() {
  // See p.43 of FreeRTOS for ESP32-Arduino
  // Delete self (main task)
  vTaskDelete(nullptr);
}

# There are a few advantages of using LittleFS over SPIFFS:
#
# LittleFS is optimized for low resource usage and it employs a wear-leveling algorithm that evenly distributes writes across the flash memory, prolonging its lifespan.
# LittleFS provides faster mount times and file access by utilizing a directory indexing structure.
# LittleFS minimizes the risk of data corruption during power loss or system failures.
# LittleFS is under active development.

all:
	$(Q)echo "Compiling $(PROJECT_NAME)..."
	$(Q)mkdir -p out
	$(Q)arduino-cli compile \
		$(VERBOSE) \
		$(CLEAN) \
		--fqbn $(BOARD) \
		--board-options "$(BOARD_OPTIONS)" \
		--output-dir out \
		$(PROJECT_NAME).ino
	$(MAKE) til
	$(MAKE) til_copy


clean:
	$(RM) -r out
	$(RM) -r til

Update config, toolchain, libraries

.PHONY: update
update:
	$(Q)echo create config at /home/geraldo/.arduino15/arduino-cli.yaml
	$(Q)arduino-cli config init || echo Config file already exists
	$(Q)arduino-cli config set network.connection_timeout 600s
	$(Q)arduino-cli config add board_manager.additional_urls https://raw.githubusercontent.com/espressif/arduino-esp32/gh-pages/package_esp32_index.json
	$(Q)cat /home/geraldo/.arduino15/arduino-cli.yaml
	$(Q)arduino-cli core update-index
	$(Q)arduino-cli core install arduino:avr
	$(Q)arduino-cli core install esp32:esp32
	$(Q)arduino-cli lib uninstall AsyncTCP             # 1.1.4
	$(Q)arduino-cli lib uninstall ESPAsyncTCP          # 
	$(Q)arduino-cli lib uninstall ESPAsyncWebServer    # ESPAsyncWebServer@3.1.0
	$(Q)arduino-cli lib install LittleFS_esp32
	$(Q)arduino-cli lib install "ESP Async TCP"        # 2.0.0
	$(Q)arduino-cli lib install "Async TCP"            # 3.4.4
	$(Q)arduino-cli lib install "ESP Async WebServer"  # ESP Async WebServer@3.7.8
	$(Q)arduino-cli lib install "ESP32 LittleFS"
	$(Q)arduino-cli upgrade

Upload binary to the board

.PHONY: upload
upload:
	$(Q)arduino-cli upload \
		$(VERBOSE) \
		$(shell pwd) \
		--input-dir out/ \
		--port $(PORT) \
		--fqbn $(BOARD) \
		--board-options "$(BOARD_OPTIONS)" \
		--verify
