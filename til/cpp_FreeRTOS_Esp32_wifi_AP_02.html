void setup() {
  Serial.begin(115200);

  delay(1000);

  // Initialize SPIFFS
  // if (!SPIFFS.begin(true)) {
  if (!SPIFFS.begin(FORMAT_SPIFFS_IF_FAILED)) {
    Serial.println("An Error has occurred while mounting SPIFFS");
    return;
  } else {
    Serial.println("SPIFFS mounted");
  }

  Serial.println("SPIFFS listing:");

  File root = SPIFFS.open("/");
  if (!root) {
    Serial.println("- failed to open directory");
    return;
  }
  if (!root.isDirectory()) {
    Serial.println(" - not a directory");
    return;
  }

  File file = root.openNextFile();
  while (file) {
    if (file.isDirectory()) {
      Serial.print("  DIR : ");
      Serial.println(file.name());
      // if (levels) {
      //   listDir(fs, file.path(), levels - 1);
      // }
    } else {
      Serial.print("  FILE: ");
      Serial.print(file.name());
      Serial.print("\tSIZE: ");
      Serial.println(file.size());
    }
    file = root.openNextFile();
  }

  WiFi.mode(WIFI_AP);

  if (WiFi.softAP(ssid, password)) {
    Serial.println("softAP Ready");
  } else {
    Serial.println("softAP Failed");
  }

  delay(1000);

  // while (WiFi.status() != WL_CONNECTED) {
  //   delay(1000);
  //   Serial.println("Connecting to WiFi..");
  //   Serial.println(WiFi.status());
  // }

  if (WiFi.softAPConfig(local_IP, gateway, subnet)) {
    Serial.println("softAPConfig Ready");
  } else {
    Serial.println("softAPConfig Failed");
  }

  IPAddress myIP = WiFi.softAPIP();

  Serial.print("SSID: ");
  Serial.println(ssid);
  Serial.print("Passworkd: ");
  Serial.println(password);
  Serial.print("AP IP address: ");
  Serial.println(myIP);

  // AsyncCallbackWebHandler& on(
  //   const char* uri,
  //   WebRequestMethodComposite method,
  //   ArRequestHandlerFunction onRequest,
  //   ArUploadHandlerFunction onUpload,
  //   ArBodyHandlerFunction onBody);

  // WebRequestMethodComposite
  //

  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request) {
    // Send file
    Serial.println("/");
    Serial.println(request->url());
    // request->send(SPIFFS, "/index.html", String(), false, processor);
    request->send(SPIFFS, "/index.html");
    Serial.println("/ END");
  });
  server.on("/style.css", HTTP_GET, [](AsyncWebServerRequest *request) {
    // Send file
    Serial.println("/style.css");
    request->send(SPIFFS, "/style.css", "text/css");
  });
  server.on("/on", HTTP_GET, [](AsyncWebServerRequest *request) {
    Serial.println("/on");
    digitalWrite(ledPin, HIGH);
    request->send(SPIFFS, "/index.html", String(), false, processor);
  });
  server.on("/off", HTTP_GET, [](AsyncWebServerRequest *request) {
    Serial.println("/off");
    digitalWrite(ledPin, LOW);
    request->send(SPIFFS, "/index.html", String(), false, processor);
  });

  server.begin();
}

all:
	$(Q)echo "Compiling $(PROJECT_NAME)..."
	$(Q)mkdir -p out
	$(Q)arduino-cli compile \
		$(VERBOSE) \
		$(CLEAN) \
		--fqbn $(BOARD) \
		--board-options "$(BOARD_OPTIONS)" \
		--output-dir out \
		$(PROJECT_NAME).ino
	$(MAKE) til
	$(MAKE) til_copy


clean:
	$(RM) -r out
	$(RM) -r til

Update config, toolchain, libraries

.PHONY: update
update:
	$(Q)echo create config at /home/geraldo/.arduino15/arduino-cli.yaml
	$(Q)arduino-cli config init || echo Config file already exists
	$(Q)arduino-cli config set network.connection_timeout 600s
	$(Q)cat /home/geraldo/.arduino15/arduino-cli.yaml
	$(Q)arduino-cli core update-index
	$(Q)arduino-cli core install arduino:avr
	$(Q)arduino-cli core install esp32:esp32
	$(Q)arduino-cli lib uninstall AsyncTCP             # 1.1.4
	$(Q)arduino-cli lib uninstall ESPAsyncTCP          # 
	$(Q)arduino-cli lib uninstall ESPAsyncWebServer    # ESPAsyncWebServer@3.1.0
	$(Q)arduino-cli lib install "ESP Async TCP"        # 2.0.0
	$(Q)arduino-cli lib install "Async TCP"            # 3.4.4
	$(Q)arduino-cli lib install "ESP Async WebServer"  # ESP Async WebServer@3.7.8
	$(Q)arduino-cli upgrade

Upload binary to the board

.PHONY: upload
upload:
	$(Q)arduino-cli upload \
		$(VERBOSE) \
		$(shell pwd) \
		--input-dir out/ \
		--port $(PORT) \
		--fqbn $(BOARD) \
		--board-options "$(BOARD_OPTIONS)" \
		--verify
