    auto print1 = [](auto value) { cout << '\n' << std::bitset<8>(value); };
    print1(value1);
    print1(value2);
    cout << "\n-------- " << label;
    print1(value3);
    cout << '\n';
  };

  auto printUnary = [](auto value1, auto value2, auto label) {

  };
